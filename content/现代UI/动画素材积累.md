# 心电图

 ```js
const step = 3;       // 相邻采样点间距：3 CSS 像素
let cursor = 0;       // 当前绘制端点，允许小数
let rate = 120;       // 每秒推进的采样点数
let lastFrame = 0;

function animate(now) {
  if (!playing) return;

  if (lastFrame) {
    cursor = (cursor + (now - lastFrame) * rate / 1000)
      % ((width - 32) / step);
  }
  lastFrame = now;
  draw();
  frameId = requestAnimationFrame(animate);
}
 ```

 每帧推进量是 经过的毫秒数 × 采样率 / 1000。取模使端点到达右边界后回到左侧，开始下一轮扫描。绘制时先清空画布，再以相同坐标重画端点之前的部分；末尾用小数部分插值，让端点平滑移动：

 ```js
function draw() {
  context.clearRect(0, 0, width, height);

  const left = 16;
  const middle = height * .54;
  const scale = height * .30 / amplitude;
  const end = Math.floor(cursor);
  const fraction = cursor - end;

  context.beginPath();
  context.moveTo(left, middle - samples[0] * scale);

  for (let index = 1; index <= end; index++) {
    context.lineTo(
      left + index * step,
      middle - samples[index % samples.length] * scale
    );
  }

  const current = samples[end % samples.length];
  const next = samples[(end + 1) % samples.length];
  const tipX = left + cursor * step;
  const tipY = middle - (current + (next - current) * fraction) * scale;
  context.lineTo(tipX, tipY);

  // 发光波形及细线
  context.shadowColor = '#82f59f';
  context.shadowBlur = 15;
  context.strokeStyle = '#54eb85';
  context.lineWidth = 3;
  context.stroke();
  context.shadowBlur = 0;
  context.strokeStyle = '#c0ffd0';
  context.lineWidth = 1.2;
  context.stroke();

  // 加粗、发光的移动端点
  context.beginPath();
  context.arc(tipX, tipY, 5, 0, Math.PI * 2);
  context.fillStyle = '#d9ffe2';
  context.shadowColor = '#82f59f';
  context.shadowBlur = 17;
  context.fill();
  context.shadowBlur = 0;
}
 ```

 index % samples.length 负责循环读取输入数组；amplitude 由数组最大绝对值计算，用于把不同幅值的数据适配到画布高度。

```dataviewjs
const canvas = dv.el('canvas', '', { attr: { width: 720, height: 240 } });
canvas.style.cssText = 'width:100%;max-width:720px;height:auto;background:#0b1a12;border-radius:12px;border:1px solid #1e3a2a;';

const context = canvas.getContext('2d');
const width = canvas.width, height = canvas.height;
const step = 3, rate = 120;
let playing = true, cursor = 0, lastFrame = 0, frameId = 0;

const samples = [];
const N = 240;
for (let i = 0; i < N; i++) {
  const t = i / N * Math.PI * 2;
  samples.push(Math.sin(t*3)*0.6 + Math.sin(t*7)*0.25 + Math.sin(t*13)*0.1);
}
const amplitude = Math.max(...samples.map(v => Math.abs(v))) || 1;

function draw() {
  context.clearRect(0, 0, width, height);
  const left = 16, middle = height * 0.54;
  const scale = height * 0.30 / amplitude;
  const end = Math.floor(cursor), fraction = cursor - end;

  context.beginPath();
  context.moveTo(left, middle - samples[0] * scale);
  for (let i = 1; i <= end; i++) {
    context.lineTo(left + i*step, middle - samples[i % samples.length] * scale);
  }
  const cur = samples[end % samples.length];
  const nxt = samples[(end + 1) % samples.length];
  const tipX = left + cursor * step;
  const tipY = middle - (cur + (nxt - cur) * fraction) * scale;
  context.lineTo(tipX, tipY);

  context.shadowColor = '#82f59f';
  context.shadowBlur = 15;
  context.strokeStyle = '#54eb85';
  context.lineWidth = 3;
  context.stroke();
  context.shadowBlur = 0;
  context.strokeStyle = '#c0ffd0';
  context.lineWidth = 1.2;
  context.stroke();

  context.beginPath();
  context.arc(tipX, tipY, 5, 0, Math.PI * 2);
  context.fillStyle = '#d9ffe2';
  context.shadowColor = '#82f59f';
  context.shadowBlur = 17;
  context.fill();
  context.shadowBlur = 0;
}

function animate(now) {
  if (!playing) return;
  if (lastFrame) {
    const span = (width - 32) / step;
    cursor = (cursor + (now - lastFrame) * rate / 1000) % span;
  }
  lastFrame = now;
  draw();
  frameId = requestAnimationFrame(animate);
}
frameId = requestAnimationFrame(animate);
```

 
# 蚂蚁爬线图

 核心是 固定连接线，只移动它上面的虚线背景，不移动整条线或模块：

 ```html
<div class="connector">
  <span class="stream"></span>
</div>
 ```

 ```css
.connector {
  position: relative;
  height: 3px;
  background: #345c49; /* 固定的底线 */
}

.stream {
  position: absolute;
  inset: -1px 0;
  background: repeating-linear-gradient(
    90deg,
    #a1f3bf 0 12px,
    transparent 12px 24px
  );
  background-size: 24px 100%;
  animation: march 700ms linear infinite;
}

@keyframes march {
  to { background-position: 24px 0; }
}
 ```

 每段亮色短划线长 12 px、间隔 12 px；背景每 700 ms 向右移动一个完整周期（24 px），视觉上就会连续“爬动”。

```dataviewjs
// 注入一次 keyframes（同一页面重复执行时不会重复添加）

  background: repeating-linear-gradient(
    90deg,
    #a1f3bf 0 12px,
    transparent 12px 24px
  );
  background-size: 24px 100%;
  animation: ant-march 700ms linear infinite;
  border-radius: 2px;
`;
connector.appendChild(stream);
```
# 柱体生长+流光强调

 核心是两段 CSS 动画：柱体只生长一次；柱体上的 ::after 充当流光，在生长结束后持续循环。不需要 JavaScript。

 ```html
<div class="plot">
  <div class="bar" style="--value: 88%; --delay: .12s"></div>
</div>
 ```

 ```css
.plot {
  border-left: 2px solid #90e5df; /* 左侧坐标轴 */
}

.bar {
  position: relative;
  width: var(--value);
  overflow: hidden;               /* 流光不越出柱体 */
  transform-origin: left center;
  animation: grow 1.15s cubic-bezier(.18, .8, .24, 1) both;
  animation-delay: var(--delay);
}

.bar::after {
  content: "";
  position: absolute;
  inset: 0 auto 0 0;
  width: 55%;
  background: linear-gradient(
    90deg, transparent, #baffdf18 22%,
    #eafff4a0 50%, #baffdf20 78%, transparent
  );
  filter: blur(7px);
  animation: sweep 4.8s linear infinite;
  animation-delay: calc(var(--delay) + 1.15s);
}

@keyframes grow {
  from { transform: scaleX(0); }
  to   { transform: scaleX(1); }
}

@keyframes sweep {
  0%        { transform: translateX(-100%); }
  75%, 100% { transform: translateX(185%); }
}
 ```

 --value 决定柱长；--delay 错开各柱入场。流光每 4.8 秒循环，前 75% 时间向右移动，后 25% 留出间隔。

```dataviewjs
// ---- 1. 注入一次 keyframes ----
if (!document.getElementById('bar-anim-style')) {
  const style = document.createElement('style');
  style.id = 'bar-anim-style';
  style.textContent = `
    @keyframes bar-grow {
      from { transform: scaleX(0); }
      to   { transform: scaleX(1); }
    }
    @keyframes bar-sweep {
      0%        { transform: translateX(-100%); }
      75%, 100% { transform: translateX(185%); }
    }
  `;
  document.head.appendChild(style);
}

// ---- 2. 数据（可替换）----
const data = [
  { label: 'A', value: 88, delay: 0.12 },
  { label: 'B', value: 62, delay: 0.24 },
  { label: 'C', value: 95, delay: 0.36 },
  { label: 'D', value: 47, delay: 0.48 },
  { label: 'E', value: 73, delay: 0.60 },
];

// ---- 3. 外层容器 ----
const plot = dv.el('div', '', { attr: { class: 'bar-plot' } });
plot.style.cssText = `
  border-left: 2px solid #90e5df;
  padding: 8px 0 8px 12px;
  display: flex;
  flex-direction: column;
  gap: 10px;
`;

// ---- 4. 逐条创建柱体 ----
data.forEach(item => {
  const row = document.createElement('div');
  row.style.cssText = 'display:flex; align-items:center; gap:10px;';

  const label = document.createElement('span');
  label.textContent = item.label;
  label.style.cssText = 'width:1.5em; color:#9fb3aa; font-size:.85em;';
  row.appendChild(label);

  const track = document.createElement('div');
  track.style.cssText = 'flex:1;';

  const bar = document.createElement('div');
  bar.style.cssText = `
    position: relative;
    height: 18px;
    width: ${item.value}%;
    overflow: hidden;
    border-radius: 4px;
    background: linear-gradient(90deg, #1e3a2a, #2d5540);
    transform-origin: left center;
    animation: bar-grow 1.15s cubic-bezier(.18,.8,.24,1) both;
    animation-delay: ${item.delay}s;
  `;

  const shine = document.createElement('span');
  shine.style.cssText = `
    position: absolute;
    inset: 0 auto 0 0;
    width: 55%;
    background: linear-gradient(
      90deg, transparent, #baffdf18 22%,
      #eafff4a0 50%, #baffdf20 78%, transparent
    );
    filter: blur(7px);
    animation: bar-sweep 4.8s linear infinite;
    animation-delay: ${item.delay + 1.15}s;
  `;
  bar.appendChild(shine);

  track.appendChild(bar);
  row.appendChild(track);

  const val = document.createElement('span');
  val.textContent = item.value + '%';
  val.style.cssText = 'width:3em; text-align:right; color:#c0ffd0; font-size:.8em;';
  row.appendChild(val);

  plot.appendChild(row);
});
```
