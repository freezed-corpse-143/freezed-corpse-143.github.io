# 心电图

 ```js
const step = 3;       // 相邻采样点间距：3 CSS 像素
let cursor = 0;       // 当前绘制端点，允许小数
let rate = 120;       // 每秒推进的采样点数
let lastFrame = 0;

function animate(now) {
  if (!playing) return;

  if (lastFrame) {
    cursor = (cursor + (now - lastFrame) * rate / 1000)
      % ((width - 32) / step);
  }
  lastFrame = now;
  draw();
  frameId = requestAnimationFrame(animate);
}
 ```

 每帧推进量是 经过的毫秒数 × 采样率 / 1000。取模使端点到达右边界后回到左侧，开始下一轮扫描。绘制时先清空画布，再以相同坐标重画端点之前的部分；末尾用小数部分插值，让端点平滑移动：

 ```js
function draw() {
  context.clearRect(0, 0, width, height);

  const left = 16;
  const middle = height * .54;
  const scale = height * .30 / amplitude;
  const end = Math.floor(cursor);
  const fraction = cursor - end;

  context.beginPath();
  context.moveTo(left, middle - samples[0] * scale);

  for (let index = 1; index <= end; index++) {
    context.lineTo(
      left + index * step,
      middle - samples[index % samples.length] * scale
    );
  }

  const current = samples[end % samples.length];
  const next = samples[(end + 1) % samples.length];
  const tipX = left + cursor * step;
  const tipY = middle - (current + (next - current) * fraction) * scale;
  context.lineTo(tipX, tipY);

  // 发光波形及细线
  context.shadowColor = '#82f59f';
  context.shadowBlur = 15;
  context.strokeStyle = '#54eb85';
  context.lineWidth = 3;
  context.stroke();
  context.shadowBlur = 0;
  context.strokeStyle = '#c0ffd0';
  context.lineWidth = 1.2;
  context.stroke();

  // 加粗、发光的移动端点
  context.beginPath();
  context.arc(tipX, tipY, 5, 0, Math.PI * 2);
  context.fillStyle = '#d9ffe2';
  context.shadowColor = '#82f59f';
  context.shadowBlur = 17;
  context.fill();
  context.shadowBlur = 0;
}
 ```

 index % samples.length 负责循环读取输入数组；amplitude 由数组最大绝对值计算，用于把不同幅值的数据适配到画布高度。

```dataviewjs
const canvas = dv.el('canvas', '', { attr: { width: 720, height: 240 } });
canvas.style.cssText = 'width:100%;max-width:720px;height:auto;background:#0b1a12;border-radius:12px;border:1px solid #1e3a2a;';

const context = canvas.getContext('2d');
const width = canvas.width, height = canvas.height;
const step = 3, rate = 120;
let playing = true, cursor = 0, lastFrame = 0, frameId = 0;

const samples = [];
const N = 240;
for (let i = 0; i < N; i++) {
  const t = i / N * Math.PI * 2;
  samples.push(Math.sin(t*3)*0.6 + Math.sin(t*7)*0.25 + Math.sin(t*13)*0.1);
}
const amplitude = Math.max(...samples.map(v => Math.abs(v))) || 1;

function draw() {
  context.clearRect(0, 0, width, height);
  const left = 16, middle = height * 0.54;
  const scale = height * 0.30 / amplitude;
  const end = Math.floor(cursor), fraction = cursor - end;

  context.beginPath();
  context.moveTo(left, middle - samples[0] * scale);
  for (let i = 1; i <= end; i++) {
    context.lineTo(left + i*step, middle - samples[i % samples.length] * scale);
  }
  const cur = samples[end % samples.length];
  const nxt = samples[(end + 1) % samples.length];
  const tipX = left + cursor * step;
  const tipY = middle - (cur + (nxt - cur) * fraction) * scale;
  context.lineTo(tipX, tipY);

  context.shadowColor = '#82f59f';
  context.shadowBlur = 15;
  context.strokeStyle = '#54eb85';
  context.lineWidth = 3;
  context.stroke();
  context.shadowBlur = 0;
  context.strokeStyle = '#c0ffd0';
  context.lineWidth = 1.2;
  context.stroke();

  context.beginPath();
  context.arc(tipX, tipY, 5, 0, Math.PI * 2);
  context.fillStyle = '#d9ffe2';
  context.shadowColor = '#82f59f';
  context.shadowBlur = 17;
  context.fill();
  context.shadowBlur = 0;
}

function animate(now) {
  if (!playing) return;
  if (lastFrame) {
    const span = (width - 32) / step;
    cursor = (cursor + (now - lastFrame) * rate / 1000) % span;
  }
  lastFrame = now;
  draw();
  frameId = requestAnimationFrame(animate);
}
frameId = requestAnimationFrame(animate);
```

# 柱体生长+流光强调

 核心是两段 CSS 动画：柱体只生长一次；柱体上的 ::after 充当流光，在生长结束后持续循环。不需要 JavaScript。

 ```html

 ```

# 神经元激活动画

核心动效由**三个部分**协同完成：光点的移动、胞体/树突的扩散、以及它们的衰减。下面按“发生顺序”逐层拆解。

---

## 一、时间轴：`position` 是唯一驱动变量

整个动画只有一个时间变量 `position`，它随帧累加：

```js
position += (time - previousTime) * speed / 2100;
```

- `position` 是**浮点数**，整数部分代表"当前在哪个神经元"，小数部分代表"在这个神经元内传播到哪一步"。
- 例如 `position = 1.4` 表示：信号已到第 2 个神经元（索引 1），并且在该神经元内走完了 40% 的流程。

**所有亮灭都通过 `age = position - i` 判断**，这是核心：

```js
const age = position - i;   // i 是神经元索引
```

- `age < 0`：信号还没到这个神经元 → 静息
- `age ≈ 0`：信号刚到达 → 最亮
- `age > 0.48`：信号已离开 → 熄灭

---

## 二、胞体与树突的扩散 + 衰退

### 1. 胞体发光：`firing`（短暂脉冲）

```js
node.classList.toggle('firing', active && age < .37);
```

- 条件：`age >= 0`（已到达）**且** `age < 0.37`
- 即胞体只亮 **0.37 个单位时间**，然后自动熄灭。
- 对应的 CSS：

```css
.neuron.firing .cell-body {
  fill: #65544c;
  stroke: #ffce83;
  filter: drop-shadow(0 0 24px #ffb267);   /* 光晕 */
}
```

### 2. 树突逐级扩散：`spread-0 ~ spread-3`

```js
for (let depth = 0; depth < 4; depth++) {
  node.classList.toggle(
    `spread-${depth}`,
    active && age >= .055 + depth * .055 && age < .48
  );
}
```

这段是**光亮扩散**的精髓，逐项看：

| 层级     | 起始 age | 说明           |
| -------- | -------- | -------------- |
| spread-0 | 0.055    | 一级树突开始亮 |
| spread-1 | 0.110    | 二级树突亮     |
| spread-2 | 0.165    | 三级树突亮     |
| spread-3 | 0.220    | 末梢树突亮     |

- **起始时间递增 0.055**：光像波浪一样从胞体向外一圈圈扩散。
- **统一在 age = 0.48 时熄灭**：越晚亮的层级亮的时间越短，形成自然的衰减。
- CSS 用 `stroke` 变金色 + `drop-shadow` 实现发光：

```css
.neuron.spread-0 .dendrite-0 { stroke: #ffd391; opacity: 1; filter: drop-shadow(0 0 5px #ffb56f); }
```

**扩散 = 逐级延迟触发；衰退 = 统一截止时间。**

---

## 三、光点移动（轴突传导）

光点位置由 `stage + travel` 决定：

```js
const stage = Math.min(Math.floor(position), count - 1);  // 当前神经元
const fraction = position - stage;                        // 段内进度
const travel = Math.max(0, Math.min(1, (fraction - .29) / .71));  // 归一化到 0~1
const x = origin + gap * (stage + travel);
```

- `fraction < 0.29`：光点还停在胞体（不移动）
- `fraction = 0.29`：光点开始沿轴突出发
- `fraction = 1.0`：到达下一个神经元
- 用 `.29` 和 `.71` 把 0~1 的进度重新映射成"停留 + 移动"的节奏。

光点发光靠 CSS：

```css
.spark {
  fill: #fff0bc;
  filter: drop-shadow(0 0 8px #ffc266) drop-shadow(0 0 19px #ffac58);
}
.spark-halo { fill: #ffc073; opacity: .12; }
```

---

## 四、轴突传导的渐进亮度

轴突上有一段"充能轨迹" `charge-track`，用 `stroke-dasharray` 动态控制长度：

```js
const length = Number(path.dataset.length);
path.style.strokeDasharray =
  `${length * Math.min(1, (fraction - .29) / .54)} ${length}`;
```

- `(fraction - .29) / .54`：把轴突传导进度归一到 0~1
- 用 dash 长度模拟"电流沿轴突前进"。

突触高亮同样是条件触发：

```js
link.classList.toggle('synapsing', conducting && fraction >= .78);
```

`fraction >= .78` 时突触末梢和终扣亮起——这就是**末梢扩散**。

---

## 五、衰退机制总结

所有"衰退"本质上都是**时间窗到期**：

| 元素            | 亮起条件                     | 熄灭条件        |
| --------------- | ---------------------------- | --------------- |
| 胞体 firing     | `age >= 0`                   | `age >= 0.37`   |
| 树突 spread-0~3 | `age >= 0.055 + depth*0.055` | `age >= 0.48`   |
| 轴突 conducting | `fraction >= .29`            | `fraction >= 1` |
| 突触 synapsing  | `fraction >= .78`            | `fraction >= 1` |

没有任何"渐隐过渡"的 JS 动画——**亮灭的平滑感全靠 CSS 的 `transition` **：

```css
.dendrite { transition: stroke .14s, opacity .14s, filter .14s; }
.cell-body { transition: stroke .15s; }
```

---

## 核心一句话

> ** `age = position - i` 决定亮不亮，`age` 落在哪个区间决定亮多久，CSS `transition` 负责让变化看起来是"扩散"和"衰退"而不是硬切换。**

如果你想调整节奏，改这几个数就够了：
- `0.37` → 胞体亮多久
- `0.055` → 树突扩散的层级间隔
- `0.48` → 树突整体衰退时间
- `.29` / `.78` → 光点出发与突触亮起的时机

// --- 核心逻辑 ---
const svg = container.querySelector('#network');
const progressBar = container.querySelector('#progress');
let position = 0;
let frame = 0;
let previousTime = 0;
const count = 2;               // 固定 2 个神经元

const cycleDuration = count - 1 + .29;  // 一轮传播的总位置长度

function makeNeuron(index) {
  let seed = (index + 1) * 2654435761 >>> 0;
  function random() {
    seed = (1664525 * seed + 1013904223) >>> 0;
    return seed / 4294967296;
  }
  const sides = 5 + (index + Math.floor(random() * 3)) % 3;
  let vertices;
  do {
    vertices = Array.from({ length: sides }, (_, k) => {
      const angle = -Math.PI / 2 + k * 2 * Math.PI / sides + (random() - .5) * .07;
      const radius = 40 + (random() - .5) * 11;
      return { x: Math.cos(angle) * radius, y: Math.sin(angle) * radius };
    });
  } while (vertices.some((b, k) => {
    const a = vertices[(k + sides - 1) % sides], c = vertices[(k + 1) % sides];
    return (b.x - a.x) * (c.y - b.y) - (b.y - a.y) * (c.x - b.x) <= 0;
  }));
  const coordinate = p => `${p.x.toFixed(1)} ${p.y.toFixed(1)}`;
  const near = (a, b) => ({ x: a.x * .87 + b.x * .13, y: a.y * .87 + b.y * .13 });
  let body = `M ${coordinate(near(vertices[0], vertices[sides - 1]))}`;
  vertices.forEach((v, k) => {
    const next = vertices[(k + 1) % sides];
    body += ` Q ${coordinate(v)} ${coordinate(near(v, next))} L ${coordinate(near(next, v))}`;
  });
  body += ' Z';
  const axon = vertices.reduce((a, b) => b.x > a.x ? b : a);
  const branches = [];
  function fork(x, y, angle, length, depth) {
    const nextAngle = angle + (random() - .5) * .25;
    const endX = x + Math.cos(nextAngle) * length;
    const endY = y + Math.sin(nextAngle) * length;
    const bend = (random() - .5) * 8;
    const midX = (x + endX) / 2 - Math.sin(nextAngle) * bend;
    const midY = (y + endY) / 2 + Math.cos(nextAngle) * bend;
    branches.push(`<path class="dendrite dendrite-${depth}" d="M ${x.toFixed(1)} ${y.toFixed(1)} Q ${midX.toFixed(1)} ${midY.toFixed(1)} ${endX.toFixed(1)} ${endY.toFixed(1)}"/>`);
    if (depth === 3) return;
    for (const direction of [-1, 1]) {
      if (depth > 0 && random() < .13 + depth * .1) continue;
      fork(endX, endY, nextAngle + direction * (.31 + random() * .28), length * (.6 + random() * .16), depth + 1);
    }
  }
  for (const vertex of vertices) {
    if (vertex === axon) continue;
    const angle = Math.atan2(vertex.y, vertex.x);
    fork(vertex.x * .94, vertex.y * .94, angle, 32 + random() * 9, 0);
  }
  return { body, branches: branches.join(''), axon };
}

function render() {
  const width = origin * 2 + gap * (count - 1);
  svg.setAttribute('viewBox', `0 0 ${width} 370`);
  svg.setAttribute('width', width);
  svg.setAttribute('height', 370);
  let shapes = `<defs><radialGradient id="cellGradient"><stop stop-color="#287386"/><stop offset="1" stop-color="#142d45"/></radialGradient></defs>`;
  for (let i = 0; i < count; i++) {
    const x = origin + gap * i;
    const neuron = makeNeuron(i);
    if (i < count - 1) {
      const axonPath = `M ${(x + neuron.axon.x).toFixed(1)} ${(y + neuron.axon.y).toFixed(1)} Q ${x + 56} ${y} ${x + 75} ${y} H ${x + 333}`;
      shapes += `<g class="axon-link" data-index="${i}"><path class="axon" d="${axonPath}"/><path class="charge-track" d="${axonPath}"/>`;
      for (let j = 0; j < 4; j++) shapes += `<rect class="sheath" x="${x + 86 + j * 65}" y="${y - 11}" width="35" height="22" rx="11"/>`;
      shapes += `<path class="axon-branch" d="M ${x + 333} ${y} Q ${x + 351} ${y - 2} ${x + 365} ${y - 24} M ${x + 350} ${y - 9} Q ${x + 356} ${y - 24} ${x + 371} ${y - 34} M ${x + 333} ${y} Q ${x + 353} ${y + 3} ${x + 370} ${y + 18} M ${x + 350} ${y + 9} Q ${x + 357} ${y + 25} ${x + 369} ${y + 30}"/>`;
      for (const [dx, dy] of [[365, -24], [371, -34], [370, 18], [369, 30]]) shapes += `<circle class="terminal" cx="${x + dx}" cy="${y + dy}" r="4"/>`;
      shapes += `<text class="transmission" x="${x + 333}" y="${y - 89}" text-anchor="middle">突触末梢</text>`;
      shapes += '</g>';
    }
    shapes += `<g class="neuron" data-index="${i}" transform="translate(${x} ${y})">
      ${neuron.branches}
      <path class="cell-ring" d="${neuron.body}" transform="scale(1.25)"/>
      <path class="cell-body" d="${neuron.body}"/>
      <ellipse class="cell-core" cx="-4" cy="-3" rx="13" ry="10" transform="rotate(-20 -4 -3)"/>
      <text class="node-index" x="0" y="148" text-anchor="middle">NEURON ${String(i + 1).padStart(2, '0')}</text>
      <text class="node-name" x="0" y="173" text-anchor="middle">神经元 ${i + 1}</text>
      ${i === 0 ? '<text class="diagram-note" x="-115" y="-132" text-anchor="middle">树突</text>' : ''}
    </g>`;
  }
  shapes += `<circle id="sparkHalo" class="spark-halo" r="27" cy="${y}" cx="${origin}" visibility="hidden"/><circle id="spark" class="spark" r="7" cy="${y}" cx="${origin}" visibility="hidden"/>`;
  svg.innerHTML = shapes;
  svg.querySelectorAll('.charge-track').forEach(path => { path.dataset.length = path.getTotalLength(); });
}

function update() {
  const stage = Math.min(Math.floor(position), count - 1);
  const fraction = position - stage;
  const travel = Math.max(0, Math.min(1, (fraction - .29) / .71));
  const x = origin + gap * (stage + (stage < count - 1 ? travel : 0));
  for (const id of ['spark', 'sparkHalo']) {
    const dot = container.querySelector(`#${id}`);
    dot.setAttribute('cx', x);
    dot.setAttribute('visibility', 'visible');
  }
  svg.querySelectorAll('.neuron').forEach((node, i) => {
    const age = position - i;
    const active = age >= 0;
    node.classList.toggle('firing', active && age < .37);
    for (let depth = 0; depth < 4; depth++) {
      node.classList.toggle(`spread-${depth}`, active && age >= .055 + depth * .055 && age < .48);
    }
  });
  svg.querySelectorAll('.axon-link').forEach((link, i) => {
    const fraction = position - i;
    const conducting = fraction >= .29 && fraction < 1;
    link.classList.toggle('conducting', conducting);
    link.classList.toggle('synapsing', conducting && fraction >= .78);
    if (conducting) {
      const path = link.querySelector('.charge-track');
      const length = Number(path.dataset.length);
      path.style.strokeDasharray = `${length * Math.min(1, (fraction - .29) / .54)} ${length}`;
    } else {
      link.querySelector('.charge-track').style.strokeDasharray = '0 99999';
    }
  });
  progressBar.style.width = `${Math.min(100, 100 * position / (count - 1))}%`;
}

function tick(time) {
  if (previousTime) {
    position += (time - previousTime) * speed / 2100;
    if (position >= cycleDuration) {
      position -= cycleDuration;
      svg.querySelectorAll('.neuron').forEach(node => {
        node.classList.remove('firing');
        for (let d = 0; d < 4; d++) node.classList.remove(`spread-${d}`);
      });
      svg.querySelectorAll('.axon-link').forEach(link => {
        link.classList.remove('conducting', 'synapsing');
      });
    }
  }
  previousTime = time;
  update();
  frame = requestAnimationFrame(tick);
}

render();
frame = requestAnimationFrame(tick);
