# 最长公共系列

代码
```python
class Solution:
	def LCS(self, s1: str, s2: str)->str:
		n=len(s1)
		m=len(s2)
		
		dp = [[0]*(m+1) for _ in range(n+1)]
		
		# 填表
		for i in range(1,n+1):
			for j in range(1, m+1):
				if s1[i-1] == s2[j-1]:
					dp[i][j] = dp[i-1][j-1] + 1
				else:
					dp[i][j] = max(dp[i-1][j], dp[i][j-1])
		
		# 回溯构造 LCS
		i, j = n, m
		res = []
		while i > 0 and j> 0:
			if s1[i-1] == s2[j-1]:
				res.append(s1[i-1])
				i -= 1
				j -= 1
			elif dp[i-1][j] >= dp[i][j-1]:
				i -= 1
			else:
				j -= 1
			if not res:
				return '-1'
			else:
				return ''.join(reversed(res))
```
